(
  function()
    ------------------------------------------------
    -- USER SETTINGS (EDIT THESE ONLY)
    ------------------------------------------------
    -- Start date/time of the first visible frame:
    local startYear  = 2026
    local startMonth = 01
    local startDay   = 21
    local startHour  = 17
    local startMin   = 01
    local startSec   = 01

    -- Timeline FPS (for TL display math)
    -- Your workflow: timeline is 60fps
    local timelineFPS = 60

    -- Timelapse mapping:
    -- TLSecondsPerFrame = how many real-world seconds each timeline frame represents
    -- For your case (1 frame = 1 real second, 60f = 1 real minute):
    local TLSecondsPerFrame = 1

    -- Calibration: value of `time` at the FIRST visible frame of this clip
    local firstFrameTime = 1

    -- Extra offset so the clock/frame start advancing on the second visible frame:
    -- (this is the +1 we discovered you needed)
    local frameAdvanceOffset = 1
    ------------------------------------------------
    -- END USER SETTINGS
    ------------------------------------------------

    -- Build start time as epoch seconds
    local start = os.time({
      year  = startYear,
      month = startMonth,
      day   = startDay,
      hour  = startHour,
      min   = startMin,
      sec   = startSec
    })

    -- Normalize `time` so first visible frame becomes frame0 = 0,
    -- then apply the +1 advance offset so the second frame starts changing.
    local frame0 = math.floor((time - firstFrameTime) + 0.5) + frameAdvanceOffset
    if frame0 < 0 then frame0 = 0 end

    -- 1-based frame number for display (Frame 1, 2, 3…)
    local frame1 = frame0 + 1

    -- Timelapse mapping:
    -- each timeline frame adds TLSecondsPerFrame seconds of real time.
    local secondsToAdd = frame0 * TLSecondsPerFrame

    local clockStr = os.date("%Y-%m-%d  %H:%M:%S", start + secondsToAdd)

    -- TL mm:ss:ff display based on timelineFPS
    local totalSeconds = math.floor(frame0 / timelineFPS)
    local ff = frame0 % timelineFPS
    local mm = math.floor(totalSeconds / 60)
    local ss = totalSeconds % 60

    local tlStr = string.format("TL %02d:%02d:%02d  Frame %d", mm, ss, ff, frame1)

    return clockStr .. "\n" .. tlStr
  end
)()
